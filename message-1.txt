local function freshState()
	local conns = {}
	return {
		alive = function() return true end,
		onCleanup = function() end,
		connect = function(sig: any, fn: any)
			local c = sig:Connect(fn)
			table.insert(conns, c)
			return c
		end,
		disconnectAll = function()
			for _, c in ipairs(conns) do
				pcall(function()
					c:Disconnect()
				end)
			end
			table.clear(conns)
		end,
	}
end

do
	local function neutralize(t: any)
		if type(t) ~= "table" then
			return
		end
		if type(t.alive) == "function" then
			pcall(function()
				t.alive = function() return false end
			end)
		end
		if type(t.disconnectAll) == "function" then
			pcall(t.disconnectAll)
		end
	end
	neutralize(getgenv().STATE)
	local ok, res = pcall(function()
		return filtergc("table", { Keys = { "alive", "onCleanup", "connect" } }, false)
	end)
	if ok and type(res) == "table" then
		for _, t in ipairs(res) do
			neutralize(t)
		end
	end
	if getgenv().HorsemanUpgrade ~= nil then
		getgenv().HorsemanUpgrade = nil
	end
	pcall(function()
		local ev = RS:FindFirstChild("SwordRemotes")
		ev = ev and ev:FindFirstChild("BaconEvent")
		local sig = ev and ev.OnClientEvent
		if sig then
			for _, c in ipairs(getconnections(sig)) do
				local f = c.Function
				if typeof(f) == "function" then
					local okS, s = pcall(function()
						return debug.info(f, "s")
					end)
					if okS and s == "" then
						pcall(function()
							c:Disconnect()
						end)
					end
				end
			end
		end
	end)
	task.wait(0.8)
end

local safeErrAt: { [string]: number } = {}
local function safeRun(label: string, fn: any)
	local ok, err = pcall(fn)
	if not ok then
		local now = os.clock()
		if not safeErrAt[label] or now - safeErrAt[label] > 30 then
			safeErrAt[label] = now
			LOG(label .. " error:", err)
		end
	end
	return ok
end

local function resetCharacter()
	if resetting then
		return
	end
	resetting = true
	LOG("resetting character to return to the plaza")
	task.wait(0.8)
	local oldChar = plr.Character
	local hum = oldChar and oldChar:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health > 0 then
		pcall(function()
			hum.Health = 0
		end)
		task.wait(0.5)
		if oldChar and oldChar.Parent and hum.Health > 0 then
			pcall(function()
				oldChar:BreakJoints()
			end)
		end
	end
	local deadline = os.clock() + 25
	while aliveOk() and os.clock() < deadline do
		local c = plr.Character
		if c and c ~= oldChar then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h and h.Health > 0 and c:FindFirstChild("HumanoidRootPart") then
				break
			end
		end
		task.wait(0.25)
	end
	task.wait(0.4)
	resetting = false
	wasInFight = false
	LOG("respawned")
end

local function enterFight(trigger: BasePart, label: string, check: () -> boolean)
	if entering or resetting or giveawayBusy then
		return
	end
	entering = true
	task.spawn(function()
		local tries = 0
		while aliveOk() and not check() and tries < 3 do
			tries += 1
			local char, hrp = getChar()
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not hrp or (hum and hum.Health <= 0) then
				break
			end
			local charAtTouch = char
			LOG(("entering %s (try %d)"):format(label, tries))
			hrp.CFrame = CFrame.new(trigger.Position)
			task.wait(0.25)
			firetouchinterest(hrp, trigger, 1)
			task.wait(0.1)
			firetouchinterest(hrp, trigger, 0)
			local deadline = os.clock() + 4
			while os.clock() < deadline and aliveOk() do
				if check() or plr.Character ~= charAtTouch then
					break
				end
				task.wait(0.2)
			end
			if check() or plr.Character ~= charAtTouch then
				LOG(label .. " session started")
				enteredGrace = os.clock()
				break
			end
			task.wait(1)
		end
		if not check() then
			LOG(label .. " entry did not start, will retry")
		end
		entering = false
	end)
end

local function switchMode(label: string)
	local newMode = label == "Giant Noob" and "noob" or (label == "Bacon Arena" and "bacon" or "halloween")
	if newMode == mode then
		return
	end
	local old = mode
	mode = newMode
	pcall(writefile, "BossFarmHubMode.txt", newMode)
	LOG("mode ->", newMode)
	task.spawn(function()
		if newMode == "bacon" and baconActive() and not inCombat() then
			return
		end
		local dirty = false
		if old == "bacon" and baconActive() then
			dirty = true
			leaveBacon()
			task.wait(1.2)
		elseif old == "noob" and inCombat() then
			dirty = true
			leaveBoss()
			task.wait(1.2)
		elseif old == "halloween" and inCombat() then
			dirty = true
		end
		if dirty or baconActive() or inCombat() then
			wasInFight = false
			resetCharacter()
		end
	end)
end

local REJOIN_MARKER = "BossFarmHubLastRejoin.txt"

local function lastRejoinAge(): number
	local ok, data = pcall(readfile, REJOIN_MARKER)
	if ok and type(data) == "string" then
		local t = tonumber(data)
		if t then
			return os.time() - t
		end
	end
	return 1e9
end

local function markRejoin()
	pcall(writefile, REJOIN_MARKER, tostring(os.time()))
end

local function pickServerJob(): string?
	local ok, res = pcall(function()
		local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Desc&limit=100"):format(game.GameId)
		local body = game:HttpGet(url)
		local data = game:GetService("HttpService"):JSONDecode(body)
		local best: string? = nil
		local bestN = 1
		if type(data) == "table" and type(data.data) == "table" then
			for _, s in ipairs(data.data) do
				local n = tonumber(s.playing) or 0
				local m = tonumber(s.maxPlayers) or 0
				if type(s.id) == "string" and s.id ~= game.JobId and n > bestN and n < m then
					best, bestN = s.id, n
				end
			end
		end
		return best
	end)
	if ok and type(res) == "string" then
		return res
	end
	return nil
end

local function doRejoin(reason: string, preferPopulated: boolean, allowFallback: boolean): boolean
	if rejoinBusy or not aliveOk() then
		return false
	end
	rejoinBusy = true
	LOG("auto rejoin: " .. reason)
	notify("Auto rejoin", reason)
	local ts = game:GetService("TeleportService")
	local job: string? = nil
	if preferPopulated then
		job = pickServerJob()
		if not job and not allowFallback then
			LOG("auto rejoin: no busier server found - staying put")
			notify("Auto rejoin", "No busier server found")
			rejoinBusy = false
			return false
		end
	end
	task.wait(1)
	markRejoin()
	local loader = [=[pcall(function()
	local Players = game:GetService("Players")
	local plr = Players.LocalPlayer
	for _ = 1, 120 do
		if plr and plr:FindFirstChild("PlayerGui") and plr.Character then
			break
		end
		task.wait(0.5)
		plr = Players.LocalPlayer
	end
	task.wait(2)
	local fn = loadstring(readfile("NoobScript.luau"))
	if fn then
		fn()
	end
end)]=]
	pcall(function()
		queue_on_teleport(loader)
	end)
	local started = false
	if job then
		started = pcall(function()
			ts:TeleportToPlaceInstance(game.PlaceId, job, plr)
		end)
		if started then
			LOG("auto rejoin: hopping to a server with players")
		end
	end
	if not started and allowFallback then
		started = pcall(function()
			ts:Teleport(game.PlaceId, plr)
		end)
	end
	if not started then
		pcall(clear_teleport_queue)
		LOG("auto rejoin: teleport failed to start")
		notify("Auto rejoin", "Teleport failed - will retry later")
		rejoinBusy = false
		return false
	end
	return true
end

local function tickFight()
	local char, hrp, hum = getChar()
	if not char or not hrp or not hum or hum.Health <= 0 then
		return
	end
	if not resetting and hum.Health < hum.MaxHealth then
		hum.Health = hum.MaxHealth
	end
	clearFrozen(char)
	local bossPos = workspace:GetAttribute("BossPos")
	if typeof(bossPos) ~= "Vector3" then
		return
	end
	local now = os.clock()
	if now - lastFollow > 0.25 then
		lastFollow = now
		local flat = Vector3.new(bossPos.X - hrp.Position.X, 0, bossPos.Z - hrp.Position.Z)
		if flat.Magnitude > 3.5 then
			hrp.CFrame = CFrame.new(bossPos.X, hrp.Position.Y, bossPos.Z)
			hrp.AssemblyLinearVelocity = Vector3.zero
		elseif flat.Magnitude > 2 then
			hum:MoveTo(hrp.Position + flat.Unit * 1)
		end
	end

	local tool, equipped = findSword(char)
	if tool and not equipped then
		pcall(function()
			hum:EquipTool(tool)
		end)
		return
	end
	local cam = workspace.CurrentCamera
	if cam then
		cam.CFrame = CFrame.lookAt(cam.CFrame.Position, bossPos)
	end
	local dir = bossPos - hrp.Position
	dir = Vector3.new(dir.X, 0, dir.Z)
	if dir.Magnitude < 0.01 then
		dir = hrp.CFrame.LookVector
	end
	if turboAttack and SwordRemotes then
		if now - lastTurbo >= TURBO_INTERVAL then
			lastTurbo = now
			pcall(function()
				SwordRemotes.Swing:FireServer(dir.Unit)
			end)
			if tool and now - lastTurboAnim > 0.7 then
				lastTurboAnim = now
				pcall(function()
					tool:Activate()
				end)
			end
		end
		return
	end
	if tool then
		pcall(function()
			tool:Activate()
		end)
		return
	end
	if now - lastSwing >= cd + 0.06 then
		lastSwing = now
		if SwordRemotes then
			pcall(function()
				SwordRemotes.Swing:FireServer(dir.Unit)
			end)
		end
	end
end

local baconTarget: BasePart? = nil
local baconTargetAt = 0
local lastFallbackScan = 0

local function findCombatant(): BasePart?
	local char, hrp = getChar()
	if not hrp then
		return nil
	end
	local best: BasePart? = nil
	local bestD: number? = nil
	for _, v in ipairs(CS:GetTagged("Combatant")) do
		if v ~= char and v:IsA("Model") then
			local r = v:FindFirstChild("HumanoidRootPart") :: BasePart?
			local h = v:FindFirstChildOfClass("Humanoid")
			if r and (not h or h.Health > 0) then
				local d = (r.Position - hrp.Position).Magnitude
				if d < 220 and (not bestD or d < bestD) then
					best, bestD = r, d
				end
			end
		end
	end
	return best
end

local function findNearbyHumanoid(): BasePart?
	local char, hrp = getChar()
	if not hrp then
		return nil
	end
	local best: BasePart? = nil
	local bestD: number? = nil
	local checks = 0
	local function consider(inst: Instance)
		if checks > 400 then
			return
		end
		checks += 1
		if inst ~= char and inst:IsA("Model") then
			local h = inst:FindFirstChildOfClass("Humanoid")
			local r: BasePart? = inst.PrimaryPart
			if not r then
				local found = inst:FindFirstChild("HumanoidRootPart") or inst:FindFirstChildWhichIsA("BasePart")
				if found and found:IsA("BasePart") then
					r = found
				end
			end
			if h and h.Health > 0 and r then
				local d = (r.Position - hrp.Position).Magnitude
				if d < 120 and (not bestD or d < bestD) then
					best, bestD = r, d
				end
			end
		end
	end
	for _, ch in ipairs(workspace:GetChildren()) do
		consider(ch)
		for _, sub in ipairs(ch:GetChildren()) do
			consider(sub)
		end
	end
	return best
end

local baconPace = 2.5

local baconRestUntil = 0
local baconPaidTotal = 0

local baconEvent = SwordRemotes and SwordRemotes:FindFirstChild("BaconEvent")
if baconEvent then
	STATE.connect(baconEvent.OnClientEvent, function(kind: any, data: any)
		if kind == "Join" then
			baconPace = 2.5
			baconUnpaidStreak = 0
			baconRestUntil = 0
			LOG("bacon: joined a fresh session - pace reset to burn the reward quota")
		elseif kind == "Kill" and type(data) == "table" then
			if data.Paid == true then
				baconUnpaidStreak = 0
				baconPaidTotal += 1
				local nowK = os.clock()
				if nowK - lastPaidEventAt > 0.5 then
					baconPace = math.max(baconPace - 0.5, 2)
					notify("Bacon reward", ("+%s Time"):format(tostring(data.Amount)))
				end
				lastPaidEventAt = nowK
				LOG(("bacon kill PAID +%s Time (pace %.1fs, total %d)"):format(tostring(data.Amount), baconPace, baconPaidTotal))
			else
				baconUnpaidStreak += 1
				baconPace = math.min(baconPace + 1, 20)
				if baconUnpaidStreak >= 15 then
					baconUnpaidStreak = 0
					baconRestUntil = os.clock() + 60
					LOG("bacon: 15 unpaid kills in a row - resting 60s to reset the reward cooldown")
				end
			end
		end
	end)
end

local function tickBacon()
	local char, hrp, hum = getChar()
	if not char or not hrp or not hum or hum.Health <= 0 then
		return
	end
	clearFrozen(char)
	local now = os.clock()
	if now - baconTargetAt > 0.5 or not (baconTarget and baconTarget.Parent) then
		baconTargetAt = now
		local t = findCombatant()
		if not t and now - lastFallbackScan > 2 then
			lastFallbackScan = now
			t = findNearbyHumanoid()
		end
		if t then
			baconTarget = t
		elseif baconTarget and not baconTarget.Parent then
			baconTarget = nil
		end
	end
	local target = baconTarget
	if not target then
		return
	end
	local flat = Vector3.new(target.Position.X - hrp.Position.X, 0, target.Position.Z - hrp.Position.Z)
	local dist = flat.Magnitude
	local dir = dist > 0.01 and flat.Unit or hrp.CFrame.LookVector
	if dist > 4.5 then
		local step = math.min(dist - 3, 2.2)
		local newPos = hrp.Position + dir * step
		hrp.CFrame = CFrame.lookAt(newPos, newPos + dir)
		hrp.AssemblyLinearVelocity = Vector3.zero
	end
	local cam = workspace.CurrentCamera
	if cam then
		cam.CFrame = CFrame.lookAt(cam.CFrame.Position, target.Position)
	end
	local canAttack = now >= baconNextAttackAt and now >= baconRestUntil
	local tool, equipped = findSword(char)
	if tool then
		if not equipped then
			pcall(function()
				hum:EquipTool(tool)
			end)
		elseif dist < 12 and canAttack then
			baconNextAttackAt = now + baconPace
			pcall(function()
				tool:Activate()
			end)
		end
		return
	end
	if dist < 12 and canAttack and now - lastSwing >= cd + 0.06 then
		lastSwing = now
		baconNextAttackAt = now + baconPace
		if SwordRemotes then
			pcall(function()
				SwordRemotes.Swing:FireServer(dir)
			end)
		end
	end
end

local CANDY_PER_TIME = 100000
pcall(function()
	local hcfg = require(RS:FindFirstChild("HalloweenConfig") :: ModuleScript)
	if type(hcfg) == "table" and type(hcfg.TimePerCandy) == "number" then

local UPGRADE_MARGIN = 1.15

local function swordDps(def: any, baseCd: number): number
	return (def.Damage or 0) / (baseCd / math.max(def.Speed or 1, 0.1))
end

local function pickUpgrade(cfg: any, owned: any, candyBal: number, timeBal: number, currentDps: number)
	local baseCd = cfg.Combat.BaseCooldown
	local candyBudget = candyBal + math.floor(timeBal / CANDY_PER_TIME)
	local best: any = nil
	for _, def in ipairs(cfg.Swords) do
		local dps = swordDps(def, baseCd)
		if not owned[def.Key] and dps > currentDps * UPGRADE_MARGIN then
			local via, price
			if type(def.Candy) == "number" and candyBudget >= def.Candy then
				via, price = "candy", def.Candy
			elseif def.Price and type(def.Price.Time) == "number" and timeBal >= def.Price.Time then
				via, price = "time", def.Price.Time
			end
			if via and (not best or dps > best.dps) then
				best = { key = def.Key, dps = dps, via = via, price = price }
			end
		end
	end
	return best
end

local function upgradeTick()
	local ok, err = pcall(function()
		local cfg = SwordConfig
		if not cfg.Swords or not swordAction then
			return
		end
		local baseCd = cfg.Combat.BaseCooldown
		local owned = {}
		for k in tostring(plr:GetAttribute("SwordsOwned") or cfg.StarterSword):gmatch("[^,]+") do
			owned[k] = true
		end
		local bestOwnedKey: string? = nil
		local bestOwnedDps = -1
		for _, def in ipairs(cfg.Swords) do
			if owned[def.Key] then
				local dps = swordDps(def, baseCd)
				if dps > bestOwnedDps then
					bestOwnedKey, bestOwnedDps = def.Key, dps
				end
			end
		end
		if not resetting and bestOwnedKey and plr:GetAttribute("SwordEquipped") ~= bestOwnedKey then
			swordAction:FireServer("Equip", bestOwnedKey)
			LOG("upgrade: equipping better owned sword " .. bestOwnedKey)
			task.wait(0.5)
		end
		local candy = tonumber(plr:GetAttribute("Candy")) or 0
		local timeBal = tonumber(plr:GetAttribute("Time")) or 0
		local best = pickUpgrade(cfg, owned, candy, timeBal, bestOwnedDps)
		if not best then
			return
		end
		LOG(("upgrade: buying %s (%.0f dps via %s, price %d)"):format(best.key, best.dps, best.via, best.price))
		notify("Upgrade", ("Buying %s (%s)"):format(best.key, best.via))
		if best.via == "candy" then
			local need = math.ceil(best.price - candy)
			while need > 0 do
				local chunk = math.min(need, 500)
				swordAction:FireServer("Convert", chunk)
				need -= chunk
				task.wait(0.9)
			end
			task.wait(0.6)
			swordAction:FireServer("ShopSword", best.key)
		else
			swordAction:FireServer("Buy", best.key, "Time")
		end
		task.wait(1)
	end)
	if not ok then
		LOG("upgrade error:", err)
	end
end

local function countList(s: any): number
	local n = 0
	for _ in tostring(s or ""):gmatch("%d+") do
		n += 1
	end
	return n
end

local playtimeFiredAt = {}
local function tryPlaytime()
	if not rewardClaim then
		return
	end
	if plr:GetAttribute("RewardsLoaded") ~= true then
		return
	end
	local start = tonumber(plr:GetAttribute("RewardsPlaytimeStart")) or 0
	local elapsed = workspace:GetServerTimeNow() - start
	local claimed = {}
	for n in tostring(plr:GetAttribute("RewardsPlaytimeClaimed") or ""):gmatch("%d+") do
		claimed[tonumber(n)] = true
	end
	for i, step in ipairs(TimeConfig.Rewards.Playtime) do
		if not claimed[i] and elapsed >= (tonumber(step.Minutes) or 0) * 60 and os.clock() - (playtimeFiredAt[i] or 0) > 45 then
			playtimeFiredAt[i] = os.clock()
			rewardClaim:FireServer("playtime", i)
			LOG(("playtime reward %d claimed (+%s time)"):format(i, tostring(step.Time)))
			notify("Rewards", ("Playtime %s min reward claimed"):format(tostring(step.Minutes)))
			task.wait(0.4)
		end
	end
end

local function ugcCurrency(item: any): (string, number)
	local cost = tonumber(item.Cost) or 0
	if cost < 1000 then
		return "candy", cost
	end
	return "time", cost
end

local function tryUgc()
	if not timeAction then
		return
	end
	local ugc = TimeConfig.UGC
	if type(ugc) ~= "table" then
		return
	end
	for i, item in ipairs(ugc) do
		if type(item) == "table" and plr:GetAttribute("Award_" .. i) ~= true then
			if not (item.VipOnly == true and plr:GetAttribute("Pass_VIP") ~= true) then
				local currency, cost = ugcCurrency(item)
				local balance = currency == "candy" and (tonumber(plr:GetAttribute("Candy")) or 0) or (tonumber(plr:GetAttribute("Time")) or 0)
				if balance >= cost then
					if currency == "candy" then
						timeAction:FireServer("BuyUGC", i)
					else
						timeAction:FireServer("ClaimUGC", i)
					end
					LOG(("UGC #%d attempt (%s %d)"):format(i, currency, cost))
					notify("UGC", ("Claiming UGC #%d (%d %s)"):format(i, cost, currency))
					task.wait(1)
				end
			end
		end
	end
end

local function doGiveaway()
	giveawayBusy = true
	local ok, err = pcall(function()
		local round = tonumber(RS:GetAttribute("GiveawayRound")) or 0
		if round <= 0 or plr:GetAttribute("GiveawayEntered") == round then
			return
		end
		local pod = workspace:FindFirstChild("GiveawayPodium")
		local podium = pod and pod:FindFirstChild("Podium")
		local pad: BasePart? = nil
		if podium then
			pad = podium:FindFirstChild("Pad") :: BasePart?
		end
		if not pad then
			return
		end
		local _, hrp, hum = getChar()
		if not hrp or not hum or hum.Health <= 0 then
			return
		end
		local prize = tonumber(RS:GetAttribute("GiveawayPrize")) or 500000
		LOG("giveaway: heading to the pad for round " .. round)
		hrp.CFrame = CFrame.new(pad.Position + Vector3.new(0, 3, 0))
		hrp.AssemblyLinearVelocity = Vector3.zero
		task.wait(0.6)
		local prompt = pad:FindFirstChild("EnterPrompt") :: ProximityPrompt?
		if prompt then
			pcall(fireproximityprompt, prompt)
		end
		task.wait(0.9)
		if plr:GetAttribute("GiveawayEntered") ~= round and giveawayEnter then
			pcall(function()
				giveawayEnter:FireServer()
			end)
			task.wait(0.9)
		end
		if plr:GetAttribute("GiveawayEntered") == round then
			LOG(("giveaway: entered round %d"):format(round))
			notify("Giveaway", ("Entered the draw for %d Time!"):format(prize))
		else
			LOG("giveaway: entry not registered yet, will retry")
		end
	end)
	if not ok then
		LOG("giveaway error:", err)
	end
	giveawayBusy = false
end

local function fmtNum(n: number): string
	return (tostring(math.floor(n)):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", ""))
end

local function clockText(d: number): string
	d = math.max(0, math.floor(d))
	if d >= 3600 then
		return ("%d:%02d:%02d"):format(d // 3600, d % 3600 // 60, d % 60)
	end
	return ("%02d:%02d"):format(d // 60, d % 60)
end

			if not autoFarm then
				stateText = "off"
			elseif mode == "bacon" then
				stateText = baconActive() and (("fighting bacons (pace %.1fs, paid %d%s)"):format(baconPace, baconPaidTotal, os.clock() < baconRestUntil and ", resting" or "")) or (entering and "entering..." or "waiting")
			elseif resetting then
				stateText = "resetting"
			elseif inCombat() then
				stateText = "fighting"
			elseif entering then
				stateText = "entering..."
			else
				stateText = "waiting"
			end
			local phase = tostring(workspace:GetAttribute("BossPhase") or "?")
			local nextAt = workspace:GetAttribute("BossNextAt")
			local noobText = "Giant Noob: " .. phase
			if phase ~= "Open" and type(nextAt) == "number" then
				noobText = noobText .. (" (next in %s)"):format(clockText(nextAt - serverNow))
			end
			local sword = tostring(plr:GetAttribute("SwordEquipped") or "?")
			local farmText = ("%s: %s | sword: %s\n%s | Time %s | Candy %d"):format(modeName, stateText, sword, noobText, fmtNum(time), candy)
			if farmText ~= lastFarm then
				lastFarm = farmText
				farmPara:SetDesc(farmText)
			end

			local dailyReady = plr:GetAttribute("RewardsDailyReady") == true
			local dailyNext = tonumber(plr:GetAttribute("RewardsDailyNext")) or 0
			local streak = tonumber(plr:GetAttribute("RewardsDailyStreak")) or 0
			local dailyText = dailyReady and "daily READY" or ("daily in " .. clockText(dailyNext - serverNow))
			local start = tonumber(plr:GetAttribute("RewardsPlaytimeStart")) or 0
			local elapsed = serverNow - start
			local claimedCount = 0
			local claimedSet = {}
			for n in tostring(plr:GetAttribute("RewardsPlaytimeClaimed") or ""):gmatch("%d+") do
				claimedSet[tonumber(n)] = true
				claimedCount += 1
			end
			local nextStep: string? = nil
			for i, step in ipairs(TimeConfig.Rewards.Playtime) do
				if not claimedSet[i] then
					local left = (tonumber(step.Minutes) or 0) * 60 - elapsed
					nextStep = left > 0 and clockText(left) or "now"
					break
				end
			end
			local freeClaimed = countList(plr:GetAttribute("SeasonFree"))
			local halloweenToday = plr:GetAttribute("HalloweenDailyDay") == math.floor(os.time() / 86400)
			local ugcClaimed = 0
			for i = 1, 8 do
				if plr:GetAttribute("Award_" .. i) == true then
					ugcClaimed += 1
				end
			end
			local rewText = ("%s | streak %d | spins %d\nplaytime %d/%d, next in %s\nseason tier %d, free claimed %d, VIP %s\nhalloween daily %s | UGC %d/8 awarded"):format(
				dailyText, streak, spins, claimedCount, #TimeConfig.Rewards.Playtime, tostring(nextStep or "-"),
				tier, freeClaimed, vip and "YES" or "NO", halloweenToday and "auto-granted today" or "pending", ugcClaimed)
			if rewText ~= lastRew then
				lastRew = rewText
				rewardsPara:SetDesc(rewText)
			end

			local round, entries, nextDraw = giveawayStatus()
			local entered = round > 0 and plr:GetAttribute("GiveawayEntered") == round
			local giveText = ("round %d | entries %d | %s | %s"):format(
				round, entries,
				nextDraw and ("draw in " .. clockText(nextDraw - serverNow)) or "draw --:--",
				entered and "ENTERED" or (autoGiveaway and "will auto-enter" or "auto off"))
			if giveText ~= lastGive then
				lastGive = giveText
				givePara:SetDesc(giveText)
			end
		end)
		task.wait(1)
	end
end)
